+    raw_report = str(report_markdown or "").strip()
+    fallback_risk_map = build_fallback_risk_map_markdown(context, language)

+    heading_patterns = (
+        r"(?mi)^\s*#{1,2}\s*Risk Map\s*$",
+        r"(?mi)^\s*#{1,2}\s*\u98ce\u9669\u5730\u56fe\s*$",

+    assessment_lens_block_pattern = re.compile(

+            raw_report[:assessment_match.end()]
+            + "\n\n"
+            + fallback_risk_map
+            + raw_report[assessment_match.end():]
